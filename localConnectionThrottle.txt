# Simulate delay and packet loss for local testing
sudo tc qdisc add dev lo root netem delay 150ms loss 5%

# Reset delay and packet loss
sudo tc qdisc del dev lo root


# FOR ACTUAL NETWORKING
# 10% packet loss
sudo tc qdisc add dev eth0 root netem loss 10%

# 150ms of ping
sudo tc qdisc add dev eth0 root netem delay 150ms

# reset 
sudo tc qdisc del dev eth0 root
